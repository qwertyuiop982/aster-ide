cmake_minimum_required(VERSION 3.22)
project(aster_runtime LANGUAGES C)
add_library(aster_runtime SHARED path_redirect.c)
target_compile_features(aster_runtime PRIVATE c_std_11)
target_compile_options(aster_runtime PRIVATE -O2 -Wall -Wextra -Werror)
target_link_options(aster_runtime PRIVATE -Wl,-z,max-page-size=16384)
target_link_libraries(aster_runtime PRIVATE dl)
