cmake_minimum_required(VERSION 3.22)
project(aster_runtime LANGUAGES CXX)
add_library(aster_runtime SHARED path_redirect.cpp)
target_compile_features(aster_runtime PRIVATE cxx_std_17)
target_compile_options(aster_runtime PRIVATE -O2 -Wall -Wextra -Werror)
target_link_options(aster_runtime PRIVATE -Wl,-z,max-page-size=16384)
add_executable(aster_redirect redirect_main.cpp)
set_target_properties(aster_redirect PROPERTIES OUTPUT_NAME libaster_redirect.so)
target_compile_features(aster_redirect PRIVATE cxx_std_17)
target_compile_options(aster_redirect PRIVATE -O2 -Wall -Wextra -Werror)
target_link_options(aster_redirect PRIVATE -Wl,-z,max-page-size=16384)
